# HP Panel Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Track current HP as in Tales of Honor, give characters names, and show an HP panel in the top right
corner to the people who control a character.

**Architecture:**
- **Rules:** pure HP functions (clamp, stage, max-HP change, bar layout) live in `src/entity.c`, so the core tests
  cover them.
- **Ops:** two new ops, `OP_SET_HEALTH` (set or change) and `OP_SET_NAME`. They go through the op layer, which
  lets owners use them, and they travel as new wire ops (protocol 5).
- **Saves:** names are saved in save format v5.
- **Panel:** `src/graphics.c` draws it from the rule functions.
- **Console:** it learns double quotes and the new `:c` options.

**Tech Stack:** C11, SDL3, Vulkan renderer (`src/draw.h`), libsodium/zstd/webp for saves, the shell e2e
tests with ImageMagick.

Spec: `docs/superpowers/specs/2026-11-25-hp-panel-design.md`.

**Conventions used in this repo:**
- Tests are run with `make test`: save_test, core_test, netmsg_test, net_test and relay_test. Each prints
  `N checks, 0 failed`.
- The e2e tests run with `make e2e` and need a display.
- Commit messages are plain sentences, with no `feat:` prefix.
- Every commit ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```c
/* ---- HP (Tales of Honor) ------------------------------------------------------------------------ */

int hp_clamp(long long hp, int max) {
    if (max < 0) max = 0;
    if (hp > max) return max;
    if (hp < -max) return -max;
    return (int)hp;
}

HpStage hp_stage(int hp, int max) {
    if (max <= 0) return HP_HEALTHY;
    long long h2 = 2LL * hp;
    if (hp <= -max) return HP_DEAD;
    if (h2 < -(long long)max) return HP_CRITICAL;
    if (hp <= 0) return HP_DYING;
    if (h2 < max) return HP_INJURED;
    return HP_HEALTHY;
}

const char *hp_stage_name(HpStage s) {
    static const char *const names[] = { "Healthy", "Injured", "Dying", "Critical", "Dead" };
    return (unsigned)s < 5 ? names[s] : "?";
}

int hp_after_max_change(int hp, int old_max, int new_max) {
    return hp >= old_max ? hp_clamp(new_max, new_max) : hp_clamp(hp, new_max);
}

HpBar hp_bar_layout(int hp, int max, float frame_x, float frame_w) {
    HpBar b = { .fill_x = frame_x };
    if (max <= 0) return b;
    double ratio = (double)hp / max;
    if (ratio > 1) ratio = 1;
    if (ratio < -1) ratio = -1;
    b.fill_w = (float)(frame_w * fabs(ratio));
    if (hp < 0) b.fill_x = frame_x - b.fill_w;
    b.lines = true;
    b.injured_x = frame_x + frame_w / 2;
    b.critical = hp <= 0;
    b.critical_x = frame_x - frame_w / 2;
    return b;
}

void entity_display_name(const Entity *e, char *out, size_t cap) {
    if (e->name[0]) snprintf(out, cap, "%s", e->name);
    else snprintf(out, cap, "Character #%u", e->id);
}
```

```c
int split_args(const char *input, char *buf, size_t cap, char **argv, int max) {
    size_t o = 0;
    int argc = 0;
    const char *p = input;
    while (*p && argc < max) {
        while (*p == ' ' || *p == '\t') p++;
        if (!*p || o + 1 >= cap) break;
        argv[argc++] = buf + o;
        bool quoted = false;
        while (*p && (quoted || (*p != ' ' && *p != '\t'))) {
            if (*p == '"') quoted = !quoted;
            else if (o + 1 < cap) buf[o++] = *p;
            p++;
        }
        buf[o++] = '\0';
    }
    return argc;
}
```

```c
bool savefmt_char_name_valid(const char *name) {
    if (!name) return false;
    size_t n = strnlen(name, SAVEFMT_CHAR_NAME_MAX);
    if (n == 0 || n >= SAVEFMT_CHAR_NAME_MAX || !utf8_valid(name, n)) return false;
    for (size_t i = 0; i < n; i++) {
        uint8_t c = (uint8_t)name[i];
        if (c < 0x20 || c == 0x7F) return false;
        if (c == 0xC2 && (uint8_t)name[i + 1] <= 0x9F) return false; /* U+0080..U+009F (utf8_valid: a continuation follows) */
    }
    return true;
}
```

```c
/* ---- HP panels (top right) -------------------------------------------------------------------------- */

#define HP_MARGIN 8

static int by_id(const void *a, const void *b) {
    uint32_t x = (*(Entity *const *)a)->id, y = (*(Entity *const *)b)->id;
    return x < y ? -1 : x > y;
}

/* A player sees their own characters, the GM (also offline) the selected ones; oldest first. */
static void hp_panel_characters(EntityList *out) {
    if (session_is_client()) {
        const char *me = session_my_name();
        for (int i = 0; i < gh->entities.count; i++) {
            Entity *e = gh->entities.items[i];
            if (e->type == ENTITY_CHARACTER && e->owner[0] && strcasecmp(e->owner, me) == 0) da_push(out, e);
        }
    } else {
        for (int i = 0; i < gh->selectedEntities.count; i++) {
            Entity *e = entity_by_id(gh->selectedEntities.items[i]);
            if (e && e->type == ENTITY_CHARACTER) da_push(out, e);
        }
    }
    if (out->count > 1) qsort(out->items, (size_t)out->count, sizeof(*out->items), by_id);
}

/* Right-aligned text at right_x on a dark background, so it reads on any map. */
static void hp_label(const char *text, float right_x, float y, float h, float fs, float ascent, float descent) {
    float w = dr_text_width(FONT_PLAIN, fs, text);
    dr_fill_rect(right_x - w - 6, y, w + 6, h, (Color){ 20, 20, 20, 200 });
    dr_text(FONT_PLAIN, fs, right_x - w - 3, y + (h + ascent - descent) / 2, COLOR_WHITE, text);
}

static void draw_hp_panels(void) {
    EntityList list = { 0 };
    hp_panel_characters(&list);
    if (list.count == 0) {
        da_free(&list);
        return;
    }
    float fs = (float)gh->console.fontSize;
    float frame_w = fs * 8, name_h = fs * 1.25f, bar_h = fs * 1.25f, panel_h = fs * 3;
    float frame_x = (float)gh->winW - HP_MARGIN - frame_w, right = frame_x + frame_w;
    float bottom = (float)(gh->winH - console_panel_rect().h);
    float ascent, descent;
    dr_font_metrics(FONT_PLAIN, fs, &ascent, &descent);

    int fit = (int)((bottom - HP_MARGIN) / panel_h);
    int shown = list.count <= fit ? list.count : (fit > 0 ? fit - 1 : 0); /* one row for "+N more" */
    const Color bg = { 20, 20, 20, 200 }, red = { 200, 30, 30, 255 }, white = COLOR_WHITE;
    float y = HP_MARGIN;
    for (int i = 0; i < shown; i++, y += panel_h) {
        const Entity *e = list.items[i];
        char name[CHAR_NAME_MAX + 16], label[CHAR_NAME_MAX + 32], hp[32];
        entity_display_name(e, name, sizeof(name));
        bool dead = hp_stage(e->health, e->maxHealth) == HP_DEAD;
        snprintf(label, sizeof(label), dead ? "%s (dead)" : "%s", name);
        hp_label(label, right, y, name_h, fs, ascent, descent);

        float by = y + name_h;
        HpBar b = hp_bar_layout(e->health, e->maxHealth, frame_x, frame_w);
        dr_fill_rect(frame_x, by, frame_w, bar_h, bg);
        if (b.fill_w > 0) dr_fill_rect(b.fill_x, by, b.fill_w, bar_h, red);
        if (b.lines) dr_line(b.injured_x, by, b.injured_x, by + bar_h, 2, white);
        if (b.critical) dr_line(b.critical_x, by - 3, b.critical_x, by + bar_h + 3, 2, white);
        dr_stroke_rect(frame_x, by, frame_w, bar_h, 1.5f, white);
        snprintf(hp, sizeof(hp), "%d/%d", e->health, e->maxHealth);
        float hw = dr_text_width(FONT_PLAIN, fs, hp);
        dr_text(FONT_PLAIN, fs, frame_x + (frame_w - hw) / 2, by + (bar_h + ascent - descent) / 2, white, hp);
    }
    if (shown < list.count) {
        char more[32];
        snprintf(more, sizeof(more), "+%d more", list.count - shown);
        hp_label(more, right, y, name_h, fs, ascent, descent);
    }
    da_free(&list);
}
```
